/* ==========================================================================
   Body Scent — script.js (ใช้ร่วมกันทุกหน้า)
   - product.html : แสดงสินค้า + กรองตาม mood
   - order.html   : เติมข้อมูลลงฟอร์มจาก URL และส่งคำสั่งซื้อ
   - admin.html   : ดึงรายการสั่งซื้อจาก Google Sheets มาแสดงเป็นตาราง
   แต่ละส่วนทำงานเฉพาะเมื่อหน้านั้นมี element ตาม id ที่กำหนด
   ========================================================================== */

(function () {
  'use strict';

  /* ------------------------------------------------------------------------
     Config
     ------------------------------------------------------------------------ */
  var PRODUCTS_URL = 'products.json';

  var APPS_SCRIPT_URL =
    'https://script.google.com/macros/s/AKfycbxcv5KDB0JHrQaB5c2ky2HpKK6rmizeRRmPU6nISV9h_8I4ITlgCUaJ7Qdvc9RljtDQ/exec';

  var SHEET_ID = '1-3jcm0As7TYGmHJKS8zWc_fx0tjHPKDmlZpUl8xIqBE';
  // ถ้าข้อมูลอยู่ในแท็บที่ไม่ใช่แท็บแรก ให้ใส่ gid ของแท็บนั้น (ดูได้ใน URL หลัง #gid=)
  var SHEET_GID = '';
  var SHEET_CSV_URL =
    'https://docs.google.com/spreadsheets/d/' + SHEET_ID + '/gviz/tq?tqx=out:csv' +
    (SHEET_GID ? '&gid=' + encodeURIComponent(SHEET_GID) : '');

  var MOODS = [
    { key: 'all', label: 'ทั้งหมด' },
    { key: 'fresh', label: 'Fresh' },
    { key: 'sweet', label: 'Sweet' },
    { key: 'confident', label: 'Confident' },
    { key: 'romance', label: 'Romance' }
  ];

  /* ------------------------------------------------------------------------
     Helpers
     ------------------------------------------------------------------------ */
  function el(tag, className, text) {
    var node = document.createElement(tag);
    if (className) node.className = className;
    if (text !== undefined && text !== null) node.textContent = text;
    return node;
  }

  function formatPrice(value) {
    var n = Number(value);
    return isNaN(n) ? String(value) : n.toLocaleString('th-TH');
  }

  function ready(fn) {
    if (document.readyState === 'loading') {
      document.addEventListener('DOMContentLoaded', fn);
    } else {
      fn();
    }
  }

  /* ========================================================================
     1) product.html
     ======================================================================== */
  function initProducts() {
    var filterBar = document.getElementById('filter-bar');
    var list = document.getElementById('product-list');
    if (!filterBar || !list) return;

    var products = [];
    var currentMood = 'all';

    function isValidMood(key) {
      return MOODS.some(function (m) { return m.key === key; });
    }

    function getMoodFromUrl() {
      var mood = (new URLSearchParams(window.location.search).get('mood') || '').toLowerCase();
      return isValidMood(mood) ? mood : 'all';
    }

    function syncUrl(mood) {
      if (!window.history || !history.replaceState) return;
      var url = new URL(window.location.href);
      if (mood === 'all') {
        url.searchParams.delete('mood');
      } else {
        url.searchParams.set('mood', mood);
      }
      history.replaceState(null, '', url.toString());
    }

    function buildOrderUrl(product) {
      var item = product.name + (product.size ? ' ' + product.size : '');
      var params = new URLSearchParams();
      params.set('item', item);
      params.set('price', product.price);
      return 'order.html?' + params.toString();
    }

    function renderFilters() {
      filterBar.textContent = '';
      MOODS.forEach(function (m) {
        var btn = el('button', 'filter__item', m.label);
        btn.type = 'button';
        btn.setAttribute('data-mood', m.key);
        btn.setAttribute('aria-pressed', String(m.key === currentMood));
        btn.addEventListener('click', function () {
          setMood(m.key);
        });
        filterBar.appendChild(btn);
      });
    }

    function createCard(product) {
      var card = el('article', 'product-card');

      var media = el('div', 'product-card__media');
      var img = el('img');
      img.src = product.image;
      img.alt = product.name;
      img.loading = 'lazy';
      media.appendChild(img);
      card.appendChild(media);

      var body = el('div', 'product-card__body');

      var moodLabel = MOODS.filter(function (m) { return m.key === product.mood; })[0];
      var mood = el('span', 'mood mood--' + product.mood);
      mood.appendChild(el('i', 'mood__dot'));
      mood.appendChild(document.createTextNode(moodLabel ? moodLabel.label : product.mood));
      body.appendChild(mood);

      body.appendChild(el('h3', 'product-card__name', product.name));

      var meta = product.size || (product.type === 'rollon' ? 'โรลออน' : '');
      if (meta) body.appendChild(el('p', 'product-card__meta', meta));

      var footer = el('div', 'product-card__footer');
      var price = el('span', 'price', formatPrice(product.price));
      price.appendChild(el('small', null, 'บาท'));
      footer.appendChild(price);

      var order = el('a', 'btn btn--accent', 'สั่งซื้อ');
      order.href = buildOrderUrl(product);
      footer.appendChild(order);

      body.appendChild(footer);
      card.appendChild(body);
      return card;
    }

    function renderProducts() {
      var shown = products.filter(function (p) {
        return currentMood === 'all' || p.mood === currentMood;
      });

      list.textContent = '';
      if (!shown.length) {
        list.appendChild(el('p', 'text-muted', 'ไม่พบสินค้าในหมวดนี้'));
        return;
      }
      shown.forEach(function (p) {
        list.appendChild(createCard(p));
      });
    }

    function setMood(mood) {
      currentMood = mood;
      Array.prototype.forEach.call(filterBar.querySelectorAll('.filter__item'), function (btn) {
        btn.setAttribute('aria-pressed', String(btn.getAttribute('data-mood') === mood));
      });
      renderProducts();
      syncUrl(mood);
    }

    currentMood = getMoodFromUrl();
    renderFilters();
    list.appendChild(el('p', 'text-muted', 'กำลังโหลดสินค้า...'));

    fetch(PRODUCTS_URL)
      .then(function (res) {
        if (!res.ok) throw new Error('HTTP ' + res.status);
        return res.json();
      })
      .then(function (data) {
        products = Array.isArray(data) ? data : [];
        renderProducts();
      })
      .catch(function (error) {
        console.error(error);
        list.textContent = '';
        list.appendChild(el('p', 'text-muted', 'โหลดข้อมูลสินค้าไม่สำเร็จ กรุณารีเฟรชหน้าอีกครั้ง'));
      });
  }

  /* ========================================================================
     2) order.html
     ======================================================================== */
  function initOrder() {
    var form = document.getElementById('orderForm');
    if (!form) return;

    var itemsInput = document.getElementById('items');
    var totalInput = document.getElementById('total');

    // เติมข้อมูลจาก URL parameter ทันทีที่โหลดหน้า (ทั้ง items และ total)
    var params = new URLSearchParams(window.location.search);
    var item = params.get('item');
    var price = params.get('price');
    if (itemsInput && item) itemsInput.value = item;
    if (totalInput && price) totalInput.value = price;

    function valueOf(id) {
      var node = document.getElementById(id);
      return node ? node.value.trim() : '';
    }

    var submitting = false;
    var submitBtn = form.querySelector('[type="submit"]');

    function setSubmitting(state) {
      submitting = state;
      if (submitBtn) submitBtn.disabled = state;
    }

    form.addEventListener('submit', function (event) {
      event.preventDefault();
      if (submitting) return;
      if (typeof form.reportValidity === 'function' && !form.reportValidity()) return;

      var payload = {
        customerName: valueOf('customerName'),
        contact: valueOf('contact'),
        items: valueOf('items'),
        total: valueOf('total'),
        note: valueOf('note')
      };

      setSubmitting(true);

      fetch(APPS_SCRIPT_URL, {
        method: 'POST',
        body: JSON.stringify(payload)
      })
        .then(() => { window.location.href = 'thankyou.html'; })
        .catch(error => {
          console.error(error);
          alert('เกิดข้อผิดพลาด กรุณาลองใหม่อีกครั้ง');
          setSubmitting(false);
        });
    });
  }

  /* ========================================================================
     3) admin.html
     ======================================================================== */

  // Parser CSV (รองรับ ", ขึ้นบรรทัดใหม่ในเครื่องหมายคำพูด, "" และ CRLF)
  function parseCSV(text) {
    var rows = [];
    var row = [];
    var field = '';
    var inQuotes = false;

    if (text.charCodeAt(0) === 0xfeff) text = text.slice(1); // ตัด BOM

    for (var i = 0; i < text.length; i++) {
      var ch = text[i];

      if (inQuotes) {
        if (ch === '"') {
          if (text[i + 1] === '"') {
            field += '"';
            i++;
          } else {
            inQuotes = false;
          }
        } else {
          field += ch;
        }
      } else if (ch === '"') {
        inQuotes = true;
      } else if (ch === ',') {
        row.push(field);
        field = '';
      } else if (ch === '\n' || ch === '\r') {
        if (ch === '\r' && text[i + 1] === '\n') i++;
        row.push(field);
        rows.push(row);
        row = [];
        field = '';
      } else {
        field += ch;
      }
    }

    if (field !== '' || row.length) {
      row.push(field);
      rows.push(row);
    }

    // ตัดแถวว่างทิ้ง
    return rows.filter(function (r) {
      return r.some(function (cell) { return cell.trim() !== ''; });
    });
  }

  function initAdmin() {
    var tbody = document.querySelector('#ordersTable tbody');
    if (!tbody) return;

    var COLUMN_COUNT = 6;

    // ชื่อหัวคอลัมน์ที่รู้จัก (ถ้าไม่ตรง จะใช้ลำดับคอลัมน์ 1-6 ตามที่กำหนด)
    var COLUMN_ALIASES = [
      ['timestamp', 'date', 'datetime', 'วันเวลา', 'วันที่', 'เวลา'],
      ['customername', 'name', 'ชื่อลูกค้า', 'ชื่อ'],
      ['contact', 'phone', 'เบอร์โทร', 'ติดต่อ', 'เบอร์โทร/line'],
      ['items', 'item', 'รายการสินค้า', 'สินค้า', 'รายการ'],
      ['total', 'price', 'จำนวนเงินรวม', 'ยอดรวม', 'ราคา'],
      ['note', 'notes', 'หมายเหตุ']
    ];

    function showMessage(message) {
      tbody.textContent = '';
      var tr = el('tr');
      var td = el('td', 'text-muted', message);
      td.colSpan = COLUMN_COUNT;
      tr.appendChild(td);
      tbody.appendChild(tr);
    }

    function resolveColumns(headerRow) {
      var header = headerRow.map(function (h) { return h.trim().toLowerCase(); });
      return COLUMN_ALIASES.map(function (aliases, position) {
        for (var i = 0; i < header.length; i++) {
          if (aliases.indexOf(header[i]) !== -1) return i;
        }
        return position;
      });
    }

    function render(rows) {
      if (rows.length < 2) {
        showMessage('ยังไม่มีรายการสั่งซื้อ');
        return;
      }

      var columns = resolveColumns(rows[0]);
      // รายการถูกเพิ่มต่อท้ายชีตเสมอ จึงกลับลำดับให้รายการล่าสุดขึ้นก่อน
      var data = rows.slice(1).reverse();

      tbody.textContent = '';
      data.forEach(function (row) {
        var tr = el('tr');
        columns.forEach(function (colIndex, i) {
          var value = (row[colIndex] || '').trim();
          if (i === 4 && value !== '') value = formatPrice(value) + ' บาท';
          tr.appendChild(el('td', null, value));
        });
        tbody.appendChild(tr);
      });
    }

    showMessage('กำลังโหลดข้อมูล...');

    fetch(SHEET_CSV_URL)
      .then(function (res) {
        if (!res.ok) throw new Error('HTTP ' + res.status);
        return res.text();
      })
      .then(function (text) {
        render(parseCSV(text));
      })
      .catch(function (error) {
        console.error(error);
        showMessage('โหลดข้อมูลไม่สำเร็จ ตรวจสอบว่าแชร์ชีตแบบ "ทุกคนที่มีลิงก์ดูได้" แล้ว');
      });
  }

  /* ------------------------------------------------------------------------
     Init
     ------------------------------------------------------------------------ */
  ready(function () {
    initProducts();
    initOrder();
    initAdmin();
  });
})();
